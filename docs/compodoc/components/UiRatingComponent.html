import { Component, OnInit } from '@angular/core';
import { RatingModule } from 'ngx-bootstrap/rating';
import { FormsModule } from '@angular/forms';


import { RatingConfig } from 'ngx-bootstrap/rating';
import { Star } from '../../../../shared/model/pages.model';
import { routes } from '../../../../shared/routes/routes';
import { RouterLink } from '@angular/router';
interface StarModel {
  show: boolean;
  half?: boolean;
  hover?: boolean;
}
/**
 * getRatingConfig function.
 * @returns {*} Result.
 */
export function getRatingConfig(): RatingConfig {
  return Object.assign(new RatingConfig(), { ariaLabel: 'My Rating' });
}
@Component({
    selector: 'app-ui-rating',
    templateUrl: './ui-rating.component.html',
    styleUrl: './ui-rating.component.scss',
    providers: [{ provide: RatingConfig, useFactory: getRatingConfig }],
    imports: [RatingModule,FormsModule,RouterLink]
})
export class UiRatingComponent implements OnInit {
  routes=routes
  max = 5;
  rate = 3;
  basicStar = [
    { show: false },
    { show: false },
    { show: false },
    { show: false },
    { show: false },
  ];
  basicStar1 = [
    { show: false },
    { show: false },
    { show: false },
    { show: false },
    { show: false },
  ];
  savedStar = [
    { show: true },
    { show: true },
    { show: true },
    { show: false },
    { show: false },
  ];
  maxStar = [
    { show: false },
    { show: false },
    { show: false },
    { show: false },
    { show: false },
  ];
  readOnlyStar = [
    { show: true },
    { show: true },
    { show: true },
    { show: true },
    { show: true },
    { show: true },
    { show: false },
    { show: false },
    { show: false },
    { show: false },
  ];
  halfRatingStar = [
    { show: true, half: false },
    { show: true, half: false },
    { show: false, half: true },
    { show: false, half: false },
    { show: false, half: false },
  ];
  heartIcon = [
    { show: false },
    { show: false },
    { show: false },
    { show: false },
    { show: false },
  ];

  /**
   * clickBasicStar function.
   * @param {*} item - Parameter.
   * @param {*} i - Parameter.
   * @param {*} array - Parameter.
   * @returns {*} Result.
   */
  clickBasicStar(item: Star, i: number, array: Star[]): void {
    if (item.show == false) {
      array.map((res: Star, index: number) => {
        if (index <= i) {
          res.show = true;
        }
      });
    } else {
      array.map((res: Star, index: number) => {
        if (index >= i && index != 0) {
          res.show = false;
        }
      });
    }

    if (array === this.halfRatingStar) {
      array.map((res: Star) => {
        res.half = false;
      });
    }
  }
  /**
   * clickBasicStar1 function.
   * @param {*} item - Parameter.
   * @param {*} i - Parameter.
   * @param {*} array - Parameter.
   * @returns {*} Result.
   */
  clickBasicStar1(item: Star, i: number, array: StarModel[]): void {
    if (item.show == false) {
      array.map((res: Star, index: number) => {
        if (index <= i) {
          res.show = true;
        }
      });
    } else {
      array.map((res: Star, index: number) => {
        if (index >= i && index != 0) {
          res.show = false;
        }
      });
    }
    this.increaseRating();
    if (array === this.halfRatingStar) {
      array.map((res: Star) => {
        res.half = false;
      });
    }
  }

  liveRating:any = 1;
  /**
   * increaseRating function.
   * @returns {*} Result.
   */
  increaseRating() {
    this.liveRating++;
  }
  /**
   * resetStars function.
   * @returns {*} Result.
   */
  resetStars(): void {
    this.heartIcon.forEach((item) => {
      item.show = false;
    });
  }

  /**
   * hoverStar1 function.
   * @param {*} isHovered - Parameter.
   * @param {*} index - Parameter.
   * @param {*} halfRatingStar - Parameter.
   * @returns {*} Result.
   */
  hoverStar1(isHovered: boolean, index: number, halfRatingStar: StarModel[]): void {
    const item = halfRatingStar[index];
    if (item && item.half) {
      item.hover = isHovered;
    }
  }
  /**
   * hoverStar function.
   * @param {*} index - Parameter.
   * @param {*} array - Parameter.
   * @returns {*} Result.
   */
  hoverStar(index: number, array: Star[]): void {
    for (let i = 0; i <= index; i++) {
      if (array[i]) {
        array[i].show = true;
      }
    }

    for (let i = index + 1; i < array.length; i++) {
      if (array[i]) {
        array[i].show = false;
      }
    }
  }
  // bread crumb items
  breadCrumbItems!: Array<object>;
  readonly = false;
  defaultSelect = 3;
  currentRate = 2;
  customColor = 4;
  hoverSelect = 2;
  hovered = 0;
  clearRate = 2;

  /**
   * constructor function.
   * @returns {*} Result.
   */
  constructor() {}

  /**
   * ngOnInit function.
   * @returns {*} Result.
   */
  ngOnInit(): void {
    this.breadCrumbItems = [
      { label: 'Advance UI' },
      { label: 'Ratings', active: true },
    ];
  }
}
