import { Component } from '@angular/core';
import { routes } from '../../../../shared/routes/routes';
import { apiResultFormat, invoiceList, pageSelection } from '../../../../shared/model/pages.model';
import { MatTableDataSource } from '@angular/material/table';
import { DataService } from '../../../../shared/data/data.service';
import { PaginationService, tablePageSize } from '../../../../shared/custom-pagination/pagination.service';
import { Router, RouterLink } from '@angular/router';
import { MatSortModule, Sort } from '@angular/material/sort';
import { CollapseHeaderComponent } from '../../../common/collapse-header/collapse-header.component';
import { MatChipsModule } from '@angular/material/chips';
import { MatSelectModule } from '@angular/material/select';
import { MatIconModule } from '@angular/material/icon';
import { DateRangePickerComponent } from '../../../common/date-range-picker/date-range-picker.component';
import { CustomPaginationComponent } from '../../../../shared/custom-pagination/custom-pagination.component';
import { FormsModule } from '@angular/forms';
import { CommonModule } from '@angular/common';
import { Editor, NgxEditorModule, Toolbar } from 'ngx-editor';
import { BsDatepickerModule } from 'ngx-bootstrap/datepicker';

@Component({
  selector: 'app-invoice-list',
  imports: [CommonModule,FormsModule,MatSortModule,BsDatepickerModule,DateRangePickerComponent,CollapseHeaderComponent,MatSelectModule,MatIconModule,MatChipsModule,RouterLink,NgxEditorModule],
  templateUrl: './invoice-list.component.html',
  styleUrl: './invoice-list.component.scss'
})
export class InvoiceListComponent {
  public routes = routes;
  text: string | undefined;
  public tableData: invoiceList[] = [];
  public pageSize = 10;
  public serialNumberArray: number[] = [];
  public totalData = 0;
  showFilter = false;
  dataSource!: MatTableDataSource<invoiceList>;
  public searchDataValue = '';
  public tableDataCopy: invoiceList[] = [];
  public actualData: invoiceList[] = [];
  bsValue = new Date();
  bsRangeValue: Date[];
  maxDate = new Date();
  /**
   * constructor function.
   * @param {*} data - Parameter.
   * @param {*} pagination - Parameter.
   * @param {*} router - Parameter.
   * @returns {*} Result.
   */
  constructor(
    private data: DataService,
    private pagination: PaginationService,
    private router: Router
  ) {
    this.data.getInvoiceList().subscribe((apiRes: apiResultFormat) => {
      this.actualData = apiRes.data;
      this.pagination.tablePageSize.subscribe((res: tablePageSize) => {
        if (this.router.url == this.routes.invoiceList) {
          this.getTableData({ skip: res.skip, limit: res.limit });
          this.pageSize = res.pageSize;
        }
      });
    });
    this.maxDate.setDate(this.maxDate.getDate() + 7);
    this.bsRangeValue = [this.bsValue, this.maxDate];
   
  }

  /**
   * getTableData function.
   * @param {*} pageOption - Parameter.
   * @returns {*} Result.
   */
  private getTableData(pageOption: pageSelection): void {
    this.data.getInvoiceList().subscribe((apiRes: apiResultFormat) => {
      this.tableData = [];
      this.tableDataCopy = [];
      this.serialNumberArray = [];
      this.totalData = apiRes.totalData;
      apiRes.data.map((res: invoiceList, index: number) => {
        const serialNumber = index + 1;
        if (index >= pageOption.skip && serialNumber <= pageOption.limit) {
          res.id = serialNumber;
          this.tableData.push(res);
          this.serialNumberArray.push(serialNumber);
          this.tableDataCopy.push(res);
        }
      });
      this.dataSource = new MatTableDataSource<invoiceList>(this.actualData);
      this.pagination.calculatePageSize.next({
        totalData: this.totalData,
        pageSize: this.pageSize,
        tableData: this.tableData,
        tableDataCopy: this.tableDataCopy,
        serialNumberArray: this.serialNumberArray,
      });
    });
  }

  /**
   * sortData function.
   * @param {*} sort - Parameter.
   * @returns {*} Result.
   */
  public sortData(sort: Sort) {
    const data = this.tableData.slice();
    if (!sort.active || sort.direction === '') {
      this.tableData = data;
    } else {
      this.tableData = data.sort((a, b) => {
        /**
         * aValue function.
         * @returns {*} Result.
         */
        const aValue = (a as never)[sort.active];
        /**
         * bValue function.
         * @returns {*} Result.
         */
        const bValue = (b as never)[sort.active];
        return (aValue < bValue ? -1 : 1) * (sort.direction === 'asc' ? 1 : -1);
      });
    }
  }
  public sidebarPopup = false;
  public sidebarPopup2 = false;
  /**
   * openSidebarPopup function.
   * @returns {*} Result.
   */
  openSidebarPopup() {
    this.sidebarPopup = !this.sidebarPopup;
  }
  /**
   * openSidebarPopup2 function.
   * @returns {*} Result.
   */
  openSidebarPopup2() {
    this.sidebarPopup2 = !this.sidebarPopup2;
  }
  /**
   * searchData function.
   * @param {*} value - Parameter.
   * @returns {*} Result.
   */
  public searchData(value: string): void {
    if (value == '') {
      this.tableData = this.tableDataCopy;
    } else {
      this.dataSource.filter = value.trim().toLowerCase();
      this.tableData = this.dataSource.filteredData;
    }
  }
  initChecked = false;
  /**
   * selectAll function.
   * @param {*} initChecked - Parameter.
   * @returns {*} Result.
   */
  selectAll(initChecked: boolean) {
    if (!initChecked) {
      this.tableData.forEach((f) => {
        f.isSelected = true;
      });
    } else {
      this.tableData.forEach((f) => {
        f.isSelected = false;
      });
    }
  }
  values1: string[] = ['Tags','Tags','Tags'];
  values2: string[] = ['Tags','Tags','Tags'];
  /**
   * onClickStar function.
   * @param {*} item - Parameter.
   * @returns {*} Result.
   */
  onClickStar(item: invoiceList) {
    item.isStarActive = !item.isStarActive;
  }
 editor!: Editor;
  editor1!: Editor;
  toolbar: Toolbar = [
    ['bold', 'italic', 'format_clear'],
    ['underline', 'strike'],
    [{ heading: ['h1', 'h2', 'h3', 'h4', 'h5', 'h6'] }],
    ['image'],
    ['link'],
  ];

/**
 * ngOnInit function.
 * @returns {*} Result.
 */
ngOnInit(): void {
    this.editor = new Editor();
    this.editor1 = new Editor();
  }
  /**
   * ngOnDestroy function.
   * @returns {*} Result.
   */
  ngOnDestroy(): void {
    this.editor.destroy();
    this.editor1.destroy();
  }
}
