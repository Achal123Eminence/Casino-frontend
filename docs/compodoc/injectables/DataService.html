  /**
   * hideAllModals function.

   */
  public hideAllModals(): void {
    const modals = document.querySelectorAll('.modal.show');
    modals.forEach((modalEl: any) => {
      const modalInstance = bootstrap.Modal.getInstance(modalEl);
      if (modalInstance) {
        modalInstance.hide();
      }
    });
  }

  /**
   * hideOffCanvasModal function.
   * @param {*} canvasId - Parameter.
   * @returns {*} Result.
   */
  public hideOffCanvasModal(canvasId: string): void {
    const element = document.getElementById(canvasId);
    if (element) {
      const bsOffcanvas = Offcanvas.getInstance(element) || new Offcanvas(element);
      bsOffcanvas.hide();
      const backdrop = document.querySelector('.offcanvas-backdrop');
      backdrop?.remove()
    }
  }

    const modalEl = document.getElementById(modalId);
